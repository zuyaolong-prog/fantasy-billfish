cmake_minimum_required(VERSION 3.20)
project(BillfishAnimationParser VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ufbx STATIC vendor/ufbx/ufbx.c)
target_include_directories(ufbx PUBLIC vendor/ufbx)
add_executable(billfish-animation-parser src/main.cpp)
target_include_directories(billfish-animation-parser PRIVATE vendor)
target_link_libraries(billfish-animation-parser PRIVATE ufbx)
add_executable(verify-real tests/verify_real.cpp)
target_include_directories(verify-real PRIVATE vendor)
target_link_libraries(verify-real PRIVATE ufbx)
add_executable(inspect-skin tests/inspect_skin.cpp)
target_include_directories(inspect-skin PRIVATE vendor)
target_link_libraries(inspect-skin PRIVATE ufbx)
if(MSVC)
 target_compile_options(billfish-animation-parser PRIVATE /W4 /utf-8 /EHsc)
 target_compile_definitions(billfish-animation-parser PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
